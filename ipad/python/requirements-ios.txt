# The Gamma server's dependencies for the iPad's embedded CPython 3.13.
# ipad/scripts/build-python.sh installs them as iOS wheels (PyPI, plus
# Flet's index at https://pypi.flet.dev, which publishes iOS builds of the
# native packages PyPI lacks). The list is backend/requirements.txt with
# the changes commented below, made only where a package has no iOS wheel
# at the server's pin. backend/tests/test_ios_requirements.py fails when the
# server gains a dependency this file does not decide on.
#
# Nothing is left out. The PDF reader, PyMuPDF, has an iOS build on Flet's
# index, so the server on the iPad reads PDFs as every other server does:
# page text with each glyph's box (quotes drawn under the words), page and
# region pictures for the AI and MCP, exported notes placed in a page's
# free space, area pictures in the Logseq export, and the base-14 fonts
# and a CJK fallback built into the library, so a PDF whose fonts are not
# embedded renders without system fonts.

fastapi==0.136.0
# Flet's newest cp313 iOS build; 3.12.0 (the server's) has none.
orjson==3.11.9
# No [standard] extras: uvloop and httptools have no iOS wheels, and the
# extras' watchfiles/pyyaml/dotenv serve only the dev reloader and config
# files. h11 and the asyncio loop are what the plain install runs on, and
# websockets (the extras' WebSocket library) is listed on its own below.
uvicorn==0.44.0
websockets==17.1
diff-match-patch==20241021
bcrypt==5.0.0
# pydantic pins its pydantic-core exactly, and Flet publishes iOS builds of
# only some cores: 2.13.4 needs pydantic-core 2.46.4, which it has (the
# server's 2.13.2 needs 2.46.2, and 2.14.x needs 2.50.x, which it has not).
pydantic==2.13.4
PyPDF2==3.0.1
# Flet's newest iOS build; the server pins 1.28.2. The wheel carries MuPDF
# as two dylibs beside the extension modules, which the app's post-build
# step turns into frameworks of their own (ipad/project.yml,
# docs/research/pymupdf-ios.md).
PyMuPDF==1.27.2.3
python-multipart==0.0.26
fractional-indexing==0.1.3
ziamath==0.13
# Flet's iOS build; the server's 50.x has none. The backend uses only
# cryptography.fernet, unchanged across these versions. The iOS build links
# OpenSSL 3 without its legacy provider and imports only with
# CRYPTOGRAPHY_OPENSSL_NO_LEGACY set, which gamma_ios.py does.
cryptography==43.0.1
mcp==1.30.0
Pillow==12.3.0
certifi==2026.07.22
